API قیمتو

دسترسی برنامه‌نویسی به داده‌های لحظه‌ای بازار ایران و جهان — REST + WebSocket

Base URL

تمام درخواست‌ها باید به این آدرس ارسال شوند.

https://api.gheymato.ir/v1

احراز هویت

برای استفاده از API باید کلید دسترسی خود را در هدر Authorization ارسال کنید. کلید خود را از پنل مدیریت دریافت کنید.

// Header
Authorization: Bearer gmk_live_xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx

Endpoints

لیست نقاط دسترسی موجود

GET /symbols دریافت لیست تمام نمادها با قیمت لحظه‌ای
GET /symbols/:id دریافت اطلاعات کامل یک نماد
GET /history/:id?period=1D&count=100 تاریخچه قیمت برای نمودار
GET /tolerance نمادهای نزدیک سقف و کف
WS /stream اتصال WebSocket برای دریافت لحظه‌ای

نمونه کد

// دریافت همه نمادها
const API_KEY = 'gmk_live_xxxxxxxx';

const res = await fetch('https://api.gheymato.ir/v1/symbols', {
  headers: {
    'Authorization': `Bearer ${API_KEY}`
  }
});

const data = await res.json();
console.log(data.gold18);
// → { price: 2357490000, changePercent: 1.54, ... }
import requests

API_KEY = 'gmk_live_xxxxxxxx'

headers = {
  'Authorization': f'Bearer {API_KEY}'
}

res = requests.get(
  'https://api.gheymato.ir/v1/symbols',
  headers=headers
)
data = res.json()

print(data['gold18']['price'])
curl -X GET "https://api.gheymato.ir/v1/symbols" \
  -H "Authorization: Bearer gmk_live_xxxxxxxx"
<?php

$API_KEY = 'gmk_live_xxxxxxxx';

$ch = curl_init();
curl_setopt($ch, CURLOPT_URL, 'https://api.gheymato.ir/v1/symbols');
curl_setopt($ch, CURLOPT_HTTPHEADER, [
  'Authorization: Bearer ' . $API_KEY
]);
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);

$response = curl_exec($ch);
$data = json_decode($response, true);

echo $data['gold18']['price'];
